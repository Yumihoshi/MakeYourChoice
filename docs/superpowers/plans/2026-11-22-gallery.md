# 艺术馆场景（Gallery）实施计划

> **面向 AI 代理的工作者**：严格按任务顺序执行，每个任务内的复选框按顺序做。步骤里给出的代码与字段值是最终实现，不是示例。场景、预制体、材质、网格等序列化资产**只能通过 Unity 编辑器或 Unity MCP 修改**，不要手写 YAML。全部任务完成后提交给用户手测，不要自己模拟操作游戏。

**目标**：把 `docs/Back D Room (2).xlsx` 的「艺术馆区域」做成可玩的独立场景 `Assets/Scenes/Gallery.unity`——玩家从泳池房地漏坠入，在近乎全黑的近雾里落地，收集三个杯子、在提交区提交，开启「白鱼的画」房间的门；房间里与豆袋交互触发「我软脚了」并做躺下运镜。设计规格（已由用户批准）见 `docs/superpowers/specs/2026-11-22-gallery-design.html`。

**架构**：网格几何在编辑期由一次性工具烘成 4 份 Mesh 资产（墙 / 地面 / 天花板 / 中心地毯），运行时只有 4 个 `MeshRenderer`，UV 按世界坐标 1 m 一平铺，中心地毯 UV 0–1 不 repeat；玩法逻辑拆成 6 个小组件（坠落入场、杯子站、拾取、提交台、画房门、豆袋），彼此只通过 Inspector 引用通信；场景切换走新增的 `SceneLoadRequest`（`LoadSceneMode.Single`，天然卸载之前所有场景）。

**技术栈**：Unity 2022.3.62f3c1 / URP 14.0.12（3D Universal Renderer）、Unity 新版 Input System、UniTask（禁协程）、DOTween Pro（`TweenAwaiter.WaitForCompletionAsync`）、Odin Inspector（`[LabelText]` 中文名）、Yarn Spinner 3.2.8（3D 飘字对话）、Unity MCP（刷新资源 / 读 Console / 改场景）。

```csharp
		/// <summary>收集满足条件的格子（x=列，z=行）。</summary>
		static HashSet<Vector2Int> Select(System.Func<char, bool> match)
		{
			var cells = new HashSet<Vector2Int>();
			for (var z = 0; z < Map.Length; z++)
			{
				for (var x = 0; x < Map[z].Length; x++)
				{
					if (match(Map[z][x]))
					{
						cells.Add(new Vector2Int(x, z));
					}
				}
			}

			return cells;
		}

		/// <summary>把格子贪心合并成矩形，减少面的数量。</summary>
		static List<RectInt> Merge(HashSet<Vector2Int> cells)
		{
			var remaining = new HashSet<Vector2Int>(cells);
			var rects = new List<RectInt>();

			while (remaining.Count > 0)
			{
				var minZ = int.MaxValue;
				var minX = int.MaxValue;
				foreach (var cell in remaining)
				{
					if (cell.y < minZ || (cell.y == minZ && cell.x < minX))
					{
						minZ = cell.y;
						minX = cell.x;
					}
				}

				var width = 0;
				while (remaining.Contains(new Vector2Int(minX + width, minZ)))
				{
					width++;
				}

				var height = 0;
				while (true)
				{
					var full = true;
					for (var x = minX; x < minX + width; x++)
					{
						if (!remaining.Contains(new Vector2Int(x, minZ + height)))
						{
							full = false;
							break;
						}
					}

					if (!full)
					{
						break;
					}

					height++;
				}

				for (var z = minZ; z < minZ + height; z++)
				{
					for (var x = minX; x < minX + width; x++)
					{
						remaining.Remove(new Vector2Int(x, z));
					}
				}

				rects.Add(new RectInt(minX, minZ, width, height));
			}

			return rects;
		}

		/// <summary>墙：每格一根 0..层高的实体柱，外加门格上方的门楣（2.44..层高）。</summary>
		static Mesh BuildWalls(List<RectInt> wallRects, List<RectInt> doorRects)
		{
			var mesh = new Mesh { name = "艺术馆墙" };
			var vertices = new List<Vector3>();
			var triangles = new List<int>();
			var uvs = new List<Vector2>();

			foreach (var rect in wallRects)
			{
				var x0 = rect.xMin;
				var x1 = rect.xMax;
				var z0 = rect.yMin;
				var z1 = rect.yMax;
				AddBox(vertices, triangles, uvs, x0, x1, z0, z1, 0f, CeilingHeight);
			}

			foreach (var rect in doorRects)
			{
				AddBox(vertices, triangles, uvs, rect.xMin, rect.xMax, rect.yMin, rect.yMax, DoorHeight, CeilingHeight);
			}

			mesh.SetVertices(vertices);
			mesh.SetTriangles(triangles, 0);
			mesh.SetUVs(0, uvs);
			mesh.RecalculateNormals();
			mesh.RecalculateBounds();
			return mesh;
		}

		/// <summary>中心红毯：一整张 8×8 的 Quad，UV 0–1 拉伸到边界，不平铺。</summary>
		static Mesh BuildRedCarpet(HashSet<Vector2Int> redCells)
		{
			var mesh = new Mesh { name = "中心地毯" };
			var vertices = new List<Vector3>();
			var triangles = new List<int>();
			var uvs = new List<Vector2>();

			var minX = int.MaxValue;
			var minZ = int.MaxValue;
			var maxX = int.MinValue;
			var maxZ = int.MinValue;
			foreach (var cell in redCells)
			{
				minX = Mathf.Min(minX, cell.x);
				minZ = Mathf.Min(minZ, cell.y);
				maxX = Mathf.Max(maxX, cell.x + 1);
				maxZ = Mathf.Max(maxZ, cell.y + 1);
			}

			AddQuad(
				vertices, triangles, uvs,
				new Vector3(minX, RedCarpetHeight, minZ), new Vector3(minX, RedCarpetHeight, maxZ),
				new Vector3(maxX, RedCarpetHeight, maxZ), new Vector3(maxX, RedCarpetHeight, minZ),
				new Vector2(0f, 0f), new Vector2(0f, 1f), new Vector2(1f, 1f), new Vector2(1f, 0f));

			mesh.SetVertices(vertices);
			mesh.SetTriangles(triangles, 0);
			mesh.SetUVs(0, uvs);
			mesh.RecalculateNormals();
			mesh.RecalculateBounds();
			return mesh;
		}

		/// <summary>写一个轴对齐盒子的六个面；UV 用世界坐标，保证 1 m 一平铺。</summary>
		static void AddBox(List<Vector3> vertices, List<int> triangles, List<Vector2> uvs,
			float x0, float x1, float z0, float z1, float y0, float y1)
		{
			AddQuad(vertices, triangles, uvs,
				new Vector3(x1, y0, z1), new Vector3(x1, y0, z0), new Vector3(x1, y1, z0), new Vector3(x1, y1, z1),
				new Vector2(z1, y0), new Vector2(z0, y0), new Vector2(z0, y1), new Vector2(z1, y1));
			AddQuad(vertices, triangles, uvs,
				new Vector3(x0, y0, z0), new Vector3(x0, y0, z1), new Vector3(x0, y1, z1), new Vector3(x0, y1, z0),
				new Vector2(z0, y0), new Vector2(z1, y0), new Vector2(z1, y1), new Vector2(z0, y1));
			AddQuad(vertices, triangles, uvs,
				new Vector3(x0, y0, z1), new Vector3(x1, y0, z1), new Vector3(x1, y1, z1), new Vector3(x0, y1, z1),
				new Vector2(x0, y0), new Vector2(x1, y0), new Vector2(x1, y1), new Vector2(x0, y1));
			AddQuad(vertices, triangles, uvs,
				new Vector3(x1, y0, z0), new Vector3(x0, y0, z0), new Vector3(x0, y1, z0), new Vector3(x1, y1, z0),

		static void AddQuad(List<Vector3> vertices, List<int> triangles, List<Vector2> uvs,
			Vector3 a, Vector3 b, Vector3 c, Vector3 d, Vector2 ua, Vector2 ub, Vector2 uc, Vector2 ud)
		{
			var start = vertices.Count;
			vertices.Add(a);
			vertices.Add(b);
			vertices.Add(c);
			vertices.Add(d);
			uvs.Add(ua);
			uvs.Add(ub);
			uvs.Add(uc);
			uvs.Add(ud);
			triangles.Add(start);
			triangles.Add(start + 1);
			triangles.Add(start + 2);
			triangles.Add(start);
			triangles.Add(start + 2);
			triangles.Add(start + 3);
		}

		static void SaveMesh(Mesh mesh, string path)
		{
			if (AssetDatabase.LoadAssetAtPath<Mesh>(path) != null)
			{
				AssetDatabase.DeleteAsset(path);
			}

			AssetDatabase.CreateAsset(mesh, path);
		}

		static Material EnsureMaterial(string path, string texturePath)
		{
			var texture = AssetDatabase.LoadAssetAtPath<Texture2D>(texturePath);
			if (texture == null)
			{
				Debug.LogError($"艺术馆材质缺少贴图：{texturePath}");
				return null;
			}

			var material = AssetDatabase.LoadAssetAtPath<Material>(path);
			if (material == null)
			{
				material = new Material(Shader.Find("Universal Render Pipeline/Lit"));
				AssetDatabase.CreateAsset(material, path);
			}

			material.SetTexture("_BaseMap", texture);
			material.SetFloat("_Smoothness", 0.05f);
			material.SetFloat("_Metallic", 0f);
			EditorUtility.SetDirty(material);
			return material;
		}

		static void Place(Transform parent, string name, string meshPath, string materialPath, bool withCollider)
		{
			var mesh = AssetDatabase.LoadAssetAtPath<Mesh>(meshPath);
			var material = AssetDatabase.LoadAssetAtPath<Material>(materialPath);
			if (mesh == null || material == null)
			{
				Debug.LogError($"艺术馆几何缺少资产：{meshPath} 或 {materialPath}，请先执行「1 生成网格资产」");
				return;
			}

			var existing = parent.Find(name);
			var go = existing != null ? existing.gameObject : new GameObject(name);
			if (existing == null)
			{
				go.transform.SetParent(parent, false);
			}

			var filter = go.GetComponent<MeshFilter>();
			if (filter == null)
			{
				filter = go.AddComponent<MeshFilter>();
			}

			filter.sharedMesh = mesh;

			var rendererComponent = go.GetComponent<MeshRenderer>();
			if (rendererComponent == null)
			{
				rendererComponent = go.AddComponent<MeshRenderer>();
			}

			rendererComponent.sharedMaterial = material;
			rendererComponent.shadowCastingMode = UnityEngine.Rendering.ShadowCastingMode.Off;

			var collider = go.GetComponent<MeshCollider>();
			if (withCollider)
			{
				collider = collider != null ? collider : go.AddComponent<MeshCollider>();
				collider.sharedMesh = mesh;
			}
			else if (collider != null)
			{
				Object.DestroyImmediate(collider);
			}
		}

		static Transform EnsurePath(string path)
		{
			var current = GameObject.Find(path);
			if (current != null)
			{
				return current.transform;
			}

			Transform parent = null;
			foreach (var segment in path.Split('/'))
			{
				var child = parent == null ? GameObject.Find(segment)?.transform : parent.Find(segment);
				if (child == null)
				{
					child = new GameObject(segment).transform;
					child.SetParent(parent, false);
				}

				parent = child;
			}

			return parent;
		}

		static void EnsureFolder(string path)
		{
			if (AssetDatabase.IsValidFolder(path))
			{
				return;
			}

			var parts = path.Split('/');
			var current = parts[0];
			for (var i = 1; i < parts.Length; i++)
			{
				var next = current + "/" + parts[i];
				if (!AssetDatabase.IsValidFolder(next))
				{
					AssetDatabase.CreateFolder(current, parts[i]);
				}

				current = next;
			}
		}
	}
}
```

```csharp
using System;
using System.Threading;
using Cysharp.Threading.Tasks;
using DG.Tweening;
using Game.Dialogue;
using Sirenix.OdinInspector;
using UnityEngine;

namespace Game.Level
{
	/// <summary>
	/// 艺术馆入场：玩家在地漏里坠落，全程只有近处能看清（线性雾拉到 3.5 m 以内），
	/// 落地后雾在 2 s 内恢复、约 1.73 s 交还控制并播落地台词。
	/// </summary>
	public sealed class GalleryEntryFall : MonoBehaviour
	{
		[SerializeField, Required, LabelText("玩家控制器")] CharacterController controller;
		[SerializeField, Required, LabelText("第一人称控制")] FirstPersonController firstPerson;
		[SerializeField, Required, LabelText("落地剧情")] DialogueTrigger landingDialogue;

		[SerializeField, LabelText("重力加速度")] float gravity = -22f;
		[SerializeField, LabelText("最大下落速度")] float maxFallSpeed = 25f;
		[SerializeField, LabelText("落地高度")] float groundY = 0.05f;
		[SerializeField, LabelText("落地超时兜底（秒）")] float fallTimeout = 3f;

		[SerializeField, LabelText("近雾起点")] float nearFogStart = 0.2f;
		[SerializeField, LabelText("近雾终点")] float nearFogEnd = 3.5f;
		[SerializeField, LabelText("正常雾起点")] float normalFogStart = 4f;
		[SerializeField, LabelText("正常雾终点")] float normalFogEnd = 30f;
		[SerializeField, LabelText("雾恢复时长")] float fogRestoreSeconds = 2f;
		[SerializeField, LabelText("交还控制延迟")] float releaseDelay = 1.73f;

		float _fallSpeed;
		float _fallTime;
		bool _landed;

		void Awake()
		{
			if (controller == null || firstPerson == null || landingDialogue == null)
			{
				Debug.LogError("GalleryEntryFall 缺少玩家控制器、第一人称控制或落地剧情引用", this);
				enabled = false;
				return;
			}

			// 雾当黑屏用：玩家此刻只能看清伸手可及的地方。
			RenderSettings.fog = true;
			RenderSettings.fogMode = FogMode.Linear;
			RenderSettings.fogColor = Color.black;
			RenderSettings.fogStartDistance = nearFogStart;
			RenderSettings.fogEndDistance = nearFogEnd;

			firstPerson.enabled = false;
			_fallSpeed = 0f;
			_fallTime = 0f;
		}

		void Update()
		{
			if (_landed)
			{
				return;
			}

			_fallTime += Time.deltaTime;
			_fallSpeed = Mathf.Max(_fallSpeed + gravity * Time.deltaTime, -maxFallSpeed);
			controller.Move(new Vector3(0f, _fallSpeed * Time.deltaTime, 0f));

			var grounded = controller.isGrounded || controller.transform.position.y <= groundY;
			if (grounded || _fallTime >= fallTimeout)
			{
				Land();
			}
		}

		void Land()
		{
			_landed = true;
			RestoreAfterLanding(destroyCancellationToken).Forget();
		}

		/// <summary>落地后把雾拉回正常值、交还控制，再播落地台词。</summary>
		async UniTaskVoid RestoreAfterLanding(CancellationToken cancellationToken)
		{
			try
			{
				var fogTween = DOTween.To(
						() => RenderSettings.fogEndDistance,
						value =>
						{
							RenderSettings.fogStartDistance = value * normalFogStart / normalFogEnd;
							RenderSettings.fogEndDistance = value;
						},
						normalFogEnd,
						fogRestoreSeconds)
					.SetEase(Ease.OutQuad);
				fogTween.WaitForCompletionAsync(cancellationToken).Forget();

				await UniTask.Delay(TimeSpan.FromSeconds(releaseDelay), DelayType.DeltaTime, cancellationToken: cancellationToken);

				firstPerson.enabled = true;
				landingDialogue.StartDialogue();
			}
			catch (OperationCanceledException)
			{
				// 场景卸载时静默退出
			}
		}
	}
}
```

```csharp
using System;
using System.Threading;
using Cysharp.Threading.Tasks;
using DG.Tweening;
using Game.Dialogue;
using QFramework;
using Sirenix.OdinInspector;
using UnityEngine;

namespace Game.Level
{
	/// <summary>
	/// 与豆袋交互：先做躺下回弹的运镜，再播「我软脚了」，台词播完后自动起身并交还控制。
	/// </summary>
	public sealed class BeanbagInteract : InteractableBase
	{
		[SerializeField, Required, LabelText("第一人称控制")] FirstPersonController firstPerson;
		[SerializeField, Required, LabelText("相机枢轴（眼睛）")] Transform lookPivot;
		[SerializeField, Required, LabelText("躺下剧情")] DialogueTrigger dialogue;
		[SerializeField, LabelText("躺下朝向锚点")] Transform roomAnchor;

		[SerializeField, LabelText("躺下时长")] float lieDownSeconds = 1.2f;
		[SerializeField, LabelText("躺下高度")] float lieHeight = 0.55f;
		[SerializeField, LabelText("躺下前移")] float lieForward = 0.25f;
		[SerializeField, LabelText("躺下仰角")] float liePitch = 55f;
		[SerializeField, LabelText("躺下侧倾")] float lieRoll = 8f;
		[SerializeField, LabelText("台词结束后停留")] float restSeconds = 0.6f;
		[SerializeField, LabelText("起身时长")] float standUpSeconds = 0.8f;

		bool _busy;

		protected override void Awake()
		{
			base.Awake();
			if (!enabled)
			{
				return;
			}

			if (firstPerson == null || lookPivot == null || dialogue == null)
			{
				Debug.LogError("BeanbagInteract 缺少第一人称控制、相机枢轴或躺下剧情引用", this);
				enabled = false;
			}
		}

		protected override bool CanInteract() => !_busy && firstPerson != null && dialogue != null && !dialogue.IsRunning;

		protected override void OnInteract() => Lie().Forget();

		async UniTaskVoid Lie()
		{
			_busy = true;
			var startPosition = lookPivot.localPosition;

			var yaw = firstPerson.transform.eulerAngles.y;
			if (roomAnchor != null)
			{
				var direction = roomAnchor.position - lookPivot.position;
				direction.y = 0f;
				if (direction.sqrMagnitude > 0.001f)
				{
					yaw = Quaternion.LookRotation(direction).eulerAngles.y;
				}
			}
			else
			{
				Debug.LogWarning("BeanbagInteract 没有指定躺下朝向锚点，改用玩家当前朝向", this);
			}

			firstPerson.enabled = false;

			var completion = new UniTaskCompletionSource();
			void OnDialogueEnded(DialogueEndedEvent ended)
			{
				if (ended.Trigger == dialogue)
				{
					completion.TrySetResult();
				}
			}

			TypeEventSystem.Global.Register<DialogueEndedEvent>(OnDialogueEnded).UnRegisterWhenGameObjectDestroyed(gameObject);

			try
			{
				var lieRotation = Quaternion.Euler(liePitch, yaw, lieRoll);
				var layDown = DOTween.Sequence();
				layDown.Append(lookPivot.DOLocalMove(new Vector3(0f, lieHeight, lieForward), lieDownSeconds).SetEase(Ease.OutBack));
				layDown.Join(lookPivot.DOLocalRotateQuaternion(lieRotation, lieDownSeconds).SetEase(Ease.OutQuad));
				await layDown.WaitForCompletionAsync(destroyCancellationToken);

				dialogue.StartDialogue();
				await completion.Task.AttachExternalCancellation(destroyCancellationToken);

				await UniTask.Delay(TimeSpan.FromSeconds(restSeconds), DelayType.DeltaTime, cancellationToken: destroyCancellationToken);

				var standUp = DOTween.Sequence();
				standUp.Append(lookPivot.DOLocalMove(startPosition, standUpSeconds).SetEase(Ease.InOutQuad));
				standUp.Join(lookPivot.DOLocalRotateQuaternion(Quaternion.Euler(firstPerson.Pitch, 0f, 0f), standUpSeconds).SetEase(Ease.InOutQuad));
				await standUp.WaitForCompletionAsync(destroyCancellationToken);

				lookPivot.localPosition = startPosition;
				lookPivot.localRotation = Quaternion.Euler(firstPerson.Pitch, 0f, 0f);
			}
			catch (OperationCanceledException)
			{
				return;
			}
			finally
			{
				if (firstPerson != null)
				{
					firstPerson.enabled = true;
				}

				_busy = false;
			}
		}
	}
}
```

## 自检

- **规格覆盖**：场景与加载（任务 2）、几何与贴图 UV（任务 1）、坠入与雾（任务 3）、两扇门与铰链方向（任务 4）、杯子与提交与门禁（任务 5）、豆袋与运镜（任务 6）、文档路由（任务 7）——规格第 1–11 节均有落点。
- **与规格的五处细化**：① 门楣改为烘进墙网格，不再单独摆物体；② Yarn 节点名用项目习惯的 `Gallery_落地` / `Gallery_豆袋`（规格写作 `落地` / `豆袋`）；③ 提交台旋转 90° 贴西墙，因为桌子进深 0.95 m，不旋转会插进 x=0..1 的墙体；④ 计数组件挂在 `= 关卡/艺术馆/流程/流程状态`（规格写作「提交区」节点，现在「提交区」只作为区域语义，提交台仍摆在 x=1 的墙边）；⑤ `锚点_豆袋` 拆成两个——`锚点_豆袋`(台词飘字位置) 与 `锚点_豆袋视线`(躺下时看的房间中心，否则锚点贴在豆袋上、玩家站得很近时朝向会退化)。
- **执行顺序的一处调整**：`GalleryCupStation` 由「任务 5」提前到「任务 4.1」创建，因为 `GalleryDoorInteract` 要引用这个类型才能编译；它的场景实例与接线仍在任务 5.3/5.5 完成。
- **类型一致性**：`GalleryDoorInteract : DoorInteract`（`base.CanInteract()` / `base.TryOpenDoor()` 均为 `protected`）；`CupPickup` / `SubmitDeskInteract` / `BeanbagInteract` 均继承 `InteractableBase`（`protected override bool CanInteract()` / `protected override void OnInteract()` / `base.Awake()`）；`BeanbagInteract` 用 `FirstPersonController.Pitch`，对应任务 6.2 的改动；`UniTaskCompletionSource` + `TypeEventSystem.Global.Register<DialogueEndedEvent>(...).UnRegisterWhenGameObjectDestroyed(gameObject)` 符合「事件订阅必须落掉」。
- **占位符扫描**：无 TODO／待定；所有坐标、角度、时长、材质路径都是终值。
